$ git init shop
Initialized empty Git repository in C:/work/shop/.git/
$ cd shop

# 새 파일은 처음에 "추적하지 않는 파일(Untracked)"이다.
$ echo 'public class Room {}' > Room.java
$ git status
On branch main

No commits yet

Untracked files:
  (use "git add <file>..." to include in what will be committed)
	Room.java

nothing added to commit but untracked files present (use "git add" to track)

# add로 스테이징하면 "커밋할 변경(Changes to be committed)"이 된다.
$ git add Room.java
$ git status
On branch main

No commits yet

Changes to be committed:
  (use "git rm --cached <file>..." to unstage)
	new file:   Room.java

$ git commit -m "Room 클래스 추가"
[main (root-commit) 10210c1] Room 클래스 추가
 1 file changed, 1 insertion(+)
 create mode 100644 Room.java

# 두 파일을 바꾸고, 하나만 골라서 커밋한다(SVN은 보통 바뀐 파일을 한꺼번에 올린다).
$ echo '// 객실 이름' >> Room.java
$ echo 'public class Reservation {}' > Reservation.java
$ git status
On branch main
Changes not staged for commit:
  (use "git add <file>..." to update what will be committed)
  (use "git restore <file>..." to discard changes in working directory)
	modified:   Room.java

Untracked files:
  (use "git add <file>..." to include in what will be committed)
	Reservation.java

no changes added to commit (use "git add" and/or "git commit -a")
# -s: 짧게 보기. 앞 글자는 스테이징 상태, 뒤 글자는 작업 폴더 상태(M 수정, ?? 추적 안 함, A 추가)
$ git status -s
 M Room.java
?? Reservation.java
$ git add Reservation.java
$ git status -s
A  Reservation.java
 M Room.java
$ git commit -m "Reservation 클래스 추가"
[main 5a8b8c2] Reservation 클래스 추가
 1 file changed, 1 insertion(+)
 create mode 100644 Reservation.java
# 스테이징한 것만 커밋됐다. Room.java 수정은 작업 폴더에 그대로 남아 있다.
$ git status -s
 M Room.java

# 차이 보기. diff는 작업 폴더와 스테이징의 차이, --staged는 스테이징과 마지막 커밋의 차이다.
$ git diff
diff --git a/Room.java b/Room.java
index 05d6331..789ebcb 100644
--- a/Room.java
+++ b/Room.java
@@ -1 +1,2 @@
 public class Room {}
+// 객실 이름
$ git add Room.java
$ git diff
$ git diff --staged
diff --git a/Room.java b/Room.java
index 05d6331..789ebcb 100644
--- a/Room.java
+++ b/Room.java
@@ -1 +1,2 @@
 public class Room {}
+// 객실 이름
$ git commit -m "Room에 주석 추가"
[main e008bc6] Room에 주석 추가
 1 file changed, 1 insertion(+)

# 이력
$ git log --oneline

$ git log --stat --oneline -2
